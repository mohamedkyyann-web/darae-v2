export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); let { url, deep, code } = req.query; if (!url) return res.status(400).json({ error: "حط رابط" }); // نظف الرابط if (!url.startsWith('http')) url = 'https://' + url; url = url.replace(/\/$/, ''); let score = 100; let vulnerabilities = []; let logs = []; let safeCount = 0; try { logs.push(`[DARAE V18 FULL + VT] فحص ${url}`); // 1- فحص أساسي - هل الموقع شغال let r = await fetch(url, { method: 'HEAD', redirect: 'follow' }).catch(()=>null); if (r) { logs.push(`[+] Status ${r.status}`); safeCount++; // 2- فحص الهيدرز الحقيقية let headers = r.headers; if (!headers.get('content-security-policy')) { vulnerabilities.push({ type: "Missing CSP", severity: "Medium", evidence: "لا يوجد CSP", fix: "أضف default-src 'self'", description: "يحمي من XSS" }); score -= 10; } else safeCount++; if (!headers.get('strict-transport-security')) { vulnerabilities.push({ type: "Missing HSTS", severity: "Medium", evidence: "بدون HSTS", fix: "أضف HSTS", description: "http ممكن" }); score -= 10; } else safeCount++; if (headers.get('server')) { vulnerabilities.push({ type: "Server Disclosure", severity: "Low", evidence: headers.get('server'), fix: "اخفي الاصدار", description: "يكشف ثغرات" }); score -= 5; } else safeCount++; if (!headers.get('x-frame-options')) { vulnerabilities.push({ type: "Clickjacking", severity: "Medium", evidence: "بدون X-Frame-Options", fix: "أضف DENY", description: "يحمي من النقر الخفي" }); score -= 10; } else safeCount++; } // 3- فحص الملفات المكشوفة (العميق فقط) if (deep === 'true' && code && code.startsWith('DARAE-')) { logs.push(`[!] فحص عميق مفعل بكود ${code}`); const sensitivePaths = [ '/.env', '/.env.bak', '/.git/', '/.git/config', '/wp-config.php', '/wp-config.php.bak', '/config.php', '/backup.zip', '/.DS_Store', '/phpinfo.php', '/admin/', '/.htaccess' ]; for (let path of sensitivePaths) { try { let check = await fetch(url + path, { method: 'HEAD' }); if (check.status === 200) { vulnerabilities.push({ type: "Exposed File", severity: "Critical", evidence: path + " مكشوف!", fix: "احذفه من public", description: `ملف حساس ${path}` }); score -= 25; logs.push(`[!] Critical: ${path} موجود!`); } } catch(e){} } logs.push("[*] بدون مفتاح VirusTotal - تخطي الفحص"); } let critical = vulnerabilities.filter(v=>v.severity==='Critical').length; let medium = vulnerabilities.filter(v=>v.severity==='Medium').length; let low = vulnerabilities.filter(v=>v.severity==='Low').length; return res.status(200).json({ url, score: Math.max(0, score), status: score > 80 ? "آمن" : score > 50 ? "متوسط" : "خطير", totalChecks: 21, realServerChecks: true, virusTotal: false, vulnerabilities, logs, safe: safeCount + (12 - vulnerabilities.length), critical, warnings: medium + low, message: deep ? "فحص عميق مكتمل" : "فحص سريع - فعل الكود للعميق" }); } catch (e) { return res.status(500).json({ error: e.message, score: 0, status: "خطأ في الفحص", logs: ["فشل الاتصال"] }); } }
